Below is a complete, **TinyForge-appropriate AssemblyScript map generator** that produces maps like the image:

- evenly distributed stars (no huge empty zones)
- short, regular lanes
- clustered feel
- connected graph with bridges
- very few dead ends
- extra loops for multiple routes
- hub + exit marking

Assumptions:

- `random(): f32` exists → returns [0,1)
- `randomRange(max: i32): i32` exists
- No dynamic allocations beyond arrays (you can later replace with static buffers)

This is written as clean reference code.

# AssemblyScript Star Lane Map Generator

## Data Structures

```ts
export class Star {
  x: i32;
  y: i32;
  degree: i32 = 0;

  isHub: bool = false;
  isExit: bool = false;

  constructor(x: i32, y: i32) {
    this.x = x;
    this.y = y;
  }
}

export class Edge {
  a: i32;
  b: i32;

  constructor(a: i32, b: i32) {
    this.a = a;
    this.b = b;
  }
}

export class MapGraph {
  stars: Array<Star> = new Array<Star>();
  edges: Array<Edge> = new Array<Edge>();
}
```

# Utility Functions

## Distance Squared (fast)

```ts
function dist2(ax: i32, ay: i32, bx: i32, by: i32): i32 {
  let dx = ax - bx;
  let dy = ay - by;
  return dx * dx + dy * dy;
}
```

---

## Edge Existence Check

```ts
function edgeExists(edges: Array<Edge>, a: i32, b: i32): bool {
  for (let i = 0; i < edges.length; i++) {
    let e = edges[i];
    if ((e.a == a && e.b == b) || (e.a == b && e.b == a)) return true;
  }
  return false;
}
```

---

## Add Edge + Update Degree

```ts
function addEdge(graph: MapGraph, a: i32, b: i32): void {
  if (a == b) return;
  if (edgeExists(graph.edges, a, b)) return;

  graph.edges.push(new Edge(a, b));
  graph.stars[a].degree++;
  graph.stars[b].degree++;
}
```

# Step 1 — Even Star Placement (Poisson-like Rejection)

This prevents empty zones and overcrowding.

```ts
function generateStars(
  count: i32,
  minDist: i32,
  width: i32,
  height: i32,
): Array<Star> {
  let stars = new Array<Star>();
  let minD2 = minDist * minDist;

  let attempts = 0;
  let maxAttempts = count * 200;

  while (stars.length < count && attempts < maxAttempts) {
    attempts++;

    let x = 20 + randomRange(width - 40);
    let y = 20 + randomRange(height - 40);

    let ok = true;

    for (let i = 0; i < stars.length; i++) {
      let s = stars[i];
      if (dist2(x, y, s.x, s.y) < minD2) {
        ok = false;
        break;
      }
    }

    if (ok) {
      stars.push(new Star(x, y));
    }
  }

  return stars;
}
```

# Step 2 — Cluster Pull (Gives Image-Style Clumps)

```ts
function applyClusterBias(
  stars: Array<Star>,
  clusterCount: i32,
  strength: f32,
): void {
  let centersX = new Array<i32>();
  let centersY = new Array<i32>();

  for (let i = 0; i < clusterCount; i++) {
    centersX.push(60 + randomRange(200));
    centersY.push(50 + randomRange(140));
  }

  for (let i = 0; i < stars.length; i++) {
    let s = stars[i];

    let c = randomRange(clusterCount);

    let dx = <f32>(centersX[c] - s.x);
    let dy = <f32>(centersY[c] - s.y);

    s.x += <i32>(dx * strength);
    s.y += <i32>(dy * strength);
  }
}
```

# Step 3 — Local Connectivity (k-Nearest Neighbors)

This avoids long MST branches.

```ts
function connectLocalNeighbors(graph: MapGraph, k: i32, maxLane: i32): void {
  let maxD2 = maxLane * maxLane;
  let n = graph.stars.length;

  for (let i = 0; i < n; i++) {
    let best = new Array<i32>();
    let bestD = new Array<i32>();

    // find k nearest neighbors
    for (let j = 0; j < n; j++) {
      if (i == j) continue;

      let d = dist2(
        graph.stars[i].x,
        graph.stars[i].y,
        graph.stars[j].x,
        graph.stars[j].y,
      );

      if (d > maxD2) continue;

      // insert sorted
      let inserted = false;
      for (let t = 0; t < best.length; t++) {
        if (d < bestD[t]) {
          best.insert(t, j);
          bestD.insert(t, d);
          inserted = true;
          break;
        }
      }

      if (!inserted && best.length < k) {
        best.push(j);
        bestD.push(d);
      }

      if (best.length > k) {
        best.pop();
        bestD.pop();
      }
    }

    // connect them
    for (let t = 0; t < best.length; t++) {
      addEdge(graph, i, best[t]);
    }
  }
}
```

# Step 4 — Ensure Full Connectivity (Bridge Components)

We connect disconnected clusters with minimal bridges.

```ts
function floodFillComponent(
  graph: MapGraph,
  start: i32,
  comp: Array<i32>,
  visited: Array<bool>,
): void {
  let stack = new Array<i32>();
  stack.push(start);
  visited[start] = true;

  while (stack.length > 0) {
    let v = stack.pop();
    comp.push(v);

    for (let i = 0; i < graph.edges.length; i++) {
      let e = graph.edges[i];

      let next = -1;
      if (e.a == v) next = e.b;
      else if (e.b == v) next = e.a;

      if (next >= 0 && !visited[next]) {
        visited[next] = true;
        stack.push(next);
      }
    }
  }
}
```

## Connect Components

```ts
function connectComponents(graph: MapGraph): void {
  let n = graph.stars.length;
  let visited = new Array<bool>(n);

  for (let i = 0; i < n; i++) visited[i] = false;

  let components = new Array<Array<i32>>();

  for (let i = 0; i < n; i++) {
    if (!visited[i]) {
      let comp = new Array<i32>();
      floodFillComponent(graph, i, comp, visited);
      components.push(comp);
    }
  }

  while (components.length > 1) {
    let cA = components[0];
    let cB = components[1];

    let bestA = -1;
    let bestB = -1;
    let bestD = 999999999;

    for (let i = 0; i < cA.length; i++) {
      for (let j = 0; j < cB.length; j++) {
        let a = cA[i];
        let b = cB[j];

        let d = dist2(
          graph.stars[a].x,
          graph.stars[a].y,
          graph.stars[b].x,
          graph.stars[b].y,
        );

        if (d < bestD) {
          bestD = d;
          bestA = a;
          bestB = b;
        }
      }
    }

    addEdge(graph, bestA, bestB);

    // merge components
    cA = cA.concat(cB);
    components.shift();
    components[0] = cA;
  }
}
```

# Step 5 — Dead-End Repair + Extra Loops

```ts
function reduceDeadEnds(graph: MapGraph): void {
  let n = graph.stars.length;

  for (let i = 0; i < n; i++) {
    if (graph.stars[i].degree == 1) {
      // connect to a second nearest node
      let best = -1;
      let bestD = 999999999;

      for (let j = 0; j < n; j++) {
        if (i == j) continue;
        if (edgeExists(graph.edges, i, j)) continue;

        let d = dist2(
          graph.stars[i].x,
          graph.stars[i].y,
          graph.stars[j].x,
          graph.stars[j].y,
        );

        if (d < bestD) {
          bestD = d;
          best = j;
        }
      }

      if (best >= 0) addEdge(graph, i, best);
    }
  }
}
```

## Add Loop Edges

```ts
function addLoops(graph: MapGraph, extra: i32): void {
  let n = graph.stars.length;

  for (let i = 0; i < extra; i++) {
    let a = randomRange(n);
    let b = randomRange(n);

    if (a == b) continue;

    // only connect if close enough
    let d = dist2(
      graph.stars[a].x,
      graph.stars[a].y,
      graph.stars[b].x,
      graph.stars[b].y,
    );

    if (d < 55 * 55) {
      addEdge(graph, a, b);
    }
  }
}
```

# Step 6 — Mark Hubs and Exits

```ts
function markHubs(graph: MapGraph, hubCount: i32): void {
  for (let h = 0; h < hubCount; h++) {
    let best = -1;
    let bestDeg = -1;

    for (let i = 0; i < graph.stars.length; i++) {
      if (graph.stars[i].isHub) continue;
      if (graph.stars[i].degree > bestDeg) {
        bestDeg = graph.stars[i].degree;
        best = i;
      }
    }

    if (best >= 0) graph.stars[best].isHub = true;
  }
}

function markExits(graph: MapGraph, exitCount: i32): void {
  for (let e = 0; e < exitCount; e++) {
    let best = -1;
    let bestScore = -1;

    for (let i = 0; i < graph.stars.length; i++) {
      if (graph.stars[i].isExit) continue;

      let s = graph.stars[i];
      let score =
        (s.x < 40 ? 1 : 0) +
        (s.x > 280 ? 1 : 0) +
        (s.y < 40 ? 1 : 0) +
        (s.y > 200 ? 1 : 0);

      if (score > bestScore) {
        bestScore = score;
        best = i;
      }
    }

    if (best >= 0) graph.stars[best].isExit = true;
  }
}
```

# Full Generator Entry Point

```ts
export function generateStarMap(): MapGraph {
  let graph = new MapGraph();

  // 1. Place stars evenly
  graph.stars = generateStars(25, 20, 320, 240);

  // 2. Cluster bias like the example image
  applyClusterBias(graph.stars, 3, 0.25);

  // 3. Local regular connections
  connectLocalNeighbors(graph, 3, 55);

  // 4. Ensure full connectivity
  connectComponents(graph);

  // 5. Remove ugly dead ends
  reduceDeadEnds(graph);

  // 6. Add loops for multiple routes
  addLoops(graph, 6);

  // 7. Mark hubs and exits
  markHubs(graph, 4);
  markExits(graph, 5);

  return graph;
}
```
